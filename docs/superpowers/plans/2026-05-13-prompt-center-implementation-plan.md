> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a first-version enterprise prompt management center with department scope, one-level categories, tags, prompt versions, publish/archive/rollback, audit logs, unified login integration, and train-exam-matching UI.

**Architecture:** Add a new independent `prompt-center` subsystem rather than extending train-exam. The backend follows the train-exam Express/MySQL/auth pattern; the frontend follows the train-exam React/Vite shell and request style; auth adds a new portal app key and authorization branch.

**Tech Stack:** Node.js, Express, MySQL, Vitest, React, Vite, Docker Compose, existing unified auth service.

```js
const mysql = require('mysql2/promise');

const DB_HOST = process.env.MYSQL_HOST || '127.0.0.1';
const DB_PORT = Number(process.env.MYSQL_PORT || 3306);
const DB_USER = process.env.MYSQL_USER || 'prompt_center_user';
const DB_PASSWORD = process.env.MYSQL_PASSWORD || 'prompt_center_pass';
const DB_NAME = process.env.MYSQL_DATABASE || 'juxin_prompt_center';
const DB_CONN_LIMIT = Number(process.env.DB_CONNECTION_LIMIT || 10);
const DB_RETRIES = Number(process.env.DB_CONNECT_RETRIES || 30);
const DB_RETRY_DELAY = Number(process.env.DB_CONNECT_DELAY_MS || 2000);

let pool;

const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

const ensureSafeIdentifier = (value, name) => {
  if (!/^[a-zA-Z0-9_]+$/.test(value || '')) {
    throw new Error(`${name} contains unsafe characters`);
  }
  return value;
};

const buildPool = ({ database, user, password } = {}) =>
  mysql.createPool({
    host: DB_HOST,
    port: DB_PORT,
    user: user || DB_USER,
    password: password !== undefined ? password : DB_PASSWORD,
    database,
    waitForConnections: true,
    connectionLimit: DB_CONN_LIMIT,
    dateStrings: true,
  });

const waitForDb = async (targetPool, label = 'database') => {
  for (let i = 0; i < DB_RETRIES; i += 1) {
    try {
      await targetPool.query('SELECT 1');
      return;
    } catch (err) {
      if (i === DB_RETRIES - 1) throw err;
      console.warn(`[db] waiting for ${label}... (${i + 1}/${DB_RETRIES})`);
      await sleep(DB_RETRY_DELAY);
    }
  }
};

const bootstrapDatabase = async () => {
  const adminUser = process.env.MYSQL_ADMIN_USER || DB_USER;
  const adminPassword =
    process.env.MYSQL_ADMIN_PASSWORD !== undefined ? process.env.MYSQL_ADMIN_PASSWORD : DB_PASSWORD;

  const safeDbName = ensureSafeIdentifier(DB_NAME, 'MYSQL_DATABASE');
  const safeAppUser = ensureSafeIdentifier(DB_USER, 'MYSQL_USER');
  const adminPool = buildPool({ user: adminUser, password: adminPassword });
  await waitForDb(adminPool, 'mysql admin connection');

  try {
    await adminPool.query(
      `CREATE DATABASE IF NOT EXISTS \`${safeDbName}\` CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci`
    );
    await adminPool.query(`CREATE USER IF NOT EXISTS '${safeAppUser}'@'%' IDENTIFIED BY ?`, [String(DB_PASSWORD)]);
    await adminPool.query(`ALTER USER '${safeAppUser}'@'%' IDENTIFIED BY ?`, [String(DB_PASSWORD)]);
    await adminPool.query(`GRANT ALL PRIVILEGES ON \`${safeDbName}\`.* TO '${safeAppUser}'@'%'`);
    await adminPool.query('FLUSH PRIVILEGES');
  } finally {
    await adminPool.end();
  }
};

const query = async (sql, params = []) => {
  const [rows] = await pool.query(sql, params);
  return rows;
};

const get = async (sql, params = []) => {
  const rows = await query(sql, params);
  return rows[0] || null;
};

const run = async (sql, params = []) => {
  const [result] = await pool.execute(sql, params);
  return result;
};

const transaction = async (fn) => {
  const conn = await pool.getConnection();
  try {
    await conn.beginTransaction();
    const tx = {
      query: async (sql, params = []) => {
        const [rows] = await conn.query(sql, params);
        return rows;
      },
      get: async (sql, params = []) => {
        const [rows] = await conn.query(sql, params);
        return rows[0] || null;
      },
      run: async (sql, params = []) => {
        const [result] = await conn.execute(sql, params);
        return result;
      },
    };
    const result = await fn(tx);
    await conn.commit();
    return result;
  } catch (err) {
    await conn.rollback();
    throw err;
  } finally {
    conn.release();
  }
};

const createSchema = async () => {
  await run(`CREATE TABLE IF NOT EXISTS prompt_departments (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(64) NOT NULL,

  await run(`CREATE TABLE IF NOT EXISTS prompt_categories (

    name VARCHAR(128) NOT NULL,
    description TEXT NULL,
    sort_order INT NOT NULL DEFAULT 0,
    is_active TINYINT NOT NULL DEFAULT 1,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    INDEX idx_prompt_categories_dept (department_code, is_active, sort_order)
  ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4`);

  await run(`CREATE TABLE IF NOT EXISTS prompts (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    department_code VARCHAR(64) NOT NULL,
    category_id BIGINT NULL,
    title VARCHAR(255) NOT NULL,
    scenario TEXT NULL,
    content MEDIUMTEXT NOT NULL,
    variables_json JSON NULL,
    tags_json JSON NULL,
    status VARCHAR(24) NOT NULL DEFAULT 'draft',
    current_version_id BIGINT NULL,
    created_by BIGINT NULL,
    created_by_name VARCHAR(128) NULL,
    updated_by BIGINT NULL,
    updated_by_name VARCHAR(128) NULL,
    published_by BIGINT NULL,
    published_by_name VARCHAR(128) NULL,
    published_at DATETIME NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  await run(`CREATE TABLE IF NOT EXISTS prompt_versions (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    prompt_id BIGINT NOT NULL,
    version_no INT NOT NULL,
    title VARCHAR(255) NOT NULL,
    scenario TEXT NULL,
```

```js
const crypto = require('crypto');

const AUTH_SERVICE_URL = process.env.AUTH_SERVICE_URL || 'http://localhost:5180';
const AUTH_SYSTEM_KEY = String(process.env.AUTH_SYSTEM_KEY || 'prompt-center').trim() || 'prompt-center';
const AUTH_COOKIE_NAME = String(process.env.AUTH_COOKIE_NAME || 'juxin_auth_token').trim() || 'juxin_auth_token';
const AUTH_FETCH_TIMEOUT_MS = Math.max(1000, Number(process.env.AUTH_FETCH_TIMEOUT_MS || 5000));

const asyncHandler = (fn) => (req, res, next) => Promise.resolve(fn(req, res, next)).catch(next);

const forbidden = (res, message = '无权限访问提示词管理中心') => res.status(403).json({ error: message });

const readBearerToken = (req) => {
  const header = String(req.headers.authorization || '').trim();
  if (header.toLowerCase().startsWith('bearer ')) return header.slice(7).trim();
  return '';
};

const readAuthToken = (req) => readBearerToken(req) || String(req.cookies?.[AUTH_COOKIE_NAME] || '').trim();

const fetchJsonWithTimeout = async (url, options = {}) => {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), AUTH_FETCH_TIMEOUT_MS);
  try {
    const res = await fetch(url, { ...options, signal: controller.signal });
    const text = await res.text();
    let data = null;
    try {
      data = text ? JSON.parse(text) : null;
    } catch {
      data = null;
    }
    if (!res.ok) {
      const err = new Error(data?.error || `统一登录请求失败(${res.status})`);
      err.status = res.status;
      throw err;
    }
    return data;
  } finally {
    clearTimeout(timer);
  }
};

const normalizeRole = (role) => String(role || '').trim().toLowerCase();
const normalizeDepartmentCode = (value) => String(value || '').trim();

const normalizeUser = (payload) => {
  const user = payload?.user || payload || {};
  const scope = payload?.scope || user.scope || {};
  return {
    id: user.id ?? user.user_id ?? null,
    username: String(user.username || user.name || '').trim(),
    role: normalizeRole(user.role),
    departmentCode: normalizeDepartmentCode(
      user.department_code || user.departmentCode || scope.department?.code || scope.department_code
    ),
    departmentName: String(user.department_name || user.departmentName || scope.department?.name || '').trim(),
  };
};

const authRequired = asyncHandler(async (req, res, next) => {
  const token = readAuthToken(req);
  if (!token) return res.status(401).json({ error: '请先登录' });
  const payload = await fetchJsonWithTimeout(`${AUTH_SERVICE_URL}/api/auth/me`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  const user = normalizeUser(payload);
  if (!user.username) return res.status(401).json({ error: '登录态无效' });
  if (user.role === 'sysadmin') return forbidden(res, '系统管理员不进入提示词业务');
  req.user = user;
  req.authToken = token;
  next();
});

const canReadAllDepartments = (user) => ['admin', 'auditor'].includes(normalizeRole(user?.role));
const canManageAllDepartments = (user) => normalizeRole(user?.role) === 'admin';
const canEditDepartment = (user, departmentCode) => {
  const role = normalizeRole(user?.role);
  if (role === 'admin') return true;
  return role === 'editor' && normalizeDepartmentCode(user?.departmentCode) === normalizeDepartmentCode(departmentCode);
};
const canPublishDepartment = (user, departmentCode) => {
  const role = normalizeRole(user?.role);
  if (role === 'admin') return true;
  return role === 'reviewer' && normalizeDepartmentCode(user?.departmentCode) === normalizeDepartmentCode(departmentCode);
};
const canReadDepartment = (user, departmentCode) => {
  if (canReadAllDepartments(user)) return true;
  return normalizeDepartmentCode(user?.departmentCode) === normalizeDepartmentCode(departmentCode);
};

const requireAdmin = (req, res, next) => {
  if (!canManageAllDepartments(req.user)) return forbidden(res);
  next();
};

const createRequestId = () => crypto.randomBytes(8).toString('hex');

module.exports = {
  AUTH_SERVICE_URL,
  AUTH_SYSTEM_KEY,
  authRequired,
  asyncHandler,
  forbidden,
  normalizeRole,
  normalizeDepartmentCode,
  canReadAllDepartments,
  canManageAllDepartments,
  canReadDepartment,
  canEditDepartment,
  canPublishDepartment,
  requireAdmin,
  createRequestId,
};
```

```js
const VARIABLE_RE = /\{\{\s*([^{}\s][^{}]*?)\s*\}\}/g;
const ALLOWED_PROMPT_STATUSES = new Set(['draft', 'published', 'archived']);

const trimString = (value) => String(value || '').trim();

const extractPromptVariables = (content) => {
  const result = [];
  const seen = new Set();
  const text = String(content || '');
  let match;
  while ((match = VARIABLE_RE.exec(text))) {
    const name = trimString(match[1]);
    if (!name || seen.has(name)) continue;
    seen.add(name);
    result.push(name);
  }
  return result;
};

const normalizeTags = (tags) => {
  const source = Array.isArray(tags)
    ? tags
    : String(tags || '')
        .split(/[，,|]/)
        .map((item) => item.trim());
  const result = [];
  const seen = new Set();
  for (const tag of source) {
    const value = trimString(tag).slice(0, 32);
    if (!value || seen.has(value)) continue;
    seen.add(value);
    result.push(value);
    if (result.length >= 12) break;
  }
  return result;
};

const normalizePromptInput = (input = {}) => {
  const title = trimString(input.title);
  const departmentCode = trimString(input.department_code || input.departmentCode);
  const categoryId = Number(input.category_id || input.categoryId || 0);
  const scenario = trimString(input.scenario);
  const content = trimString(input.content);
  const changeNote = trimString(input.change_note || input.changeNote);
  const tags = normalizeTags(input.tags || input.tags_json);
  const variables = extractPromptVariables(content).map((name) => ({ name, description: '' }));

  if (!title) throw new Error('提示词标题不能为空');
  if (!departmentCode) throw new Error('提示词部门不能为空');
  if (!Number.isInteger(categoryId) || categoryId <= 0) throw new Error('提示词分类不能为空');
  if (!content) throw new Error('提示词正文不能为空');

  return {
    title,
    departmentCode,
    categoryId,
    scenario,
    content,
    tags,
    variables,
    changeNote,
  };
};

const canTransitionPromptStatus = (from, to) => {
  const current = trimString(from) || 'draft';
  const next = trimString(to);
  if (!ALLOWED_PROMPT_STATUSES.has(current) || !ALLOWED_PROMPT_STATUSES.has(next)) return false;
  if (current === next) return true;
  if (current === 'draft' && next === 'published') return true;
  if (current === 'published' && next === 'archived') return true;
  if (current === 'draft' && next === 'archived') return true;
  return false;
};

const encodeJson = (value) => JSON.stringify(value || []);
const decodeJson = (value, fallback) => {
  if (value === null || value === undefined || value === '') return fallback;
  if (typeof value === 'object') return value;
  try {
    return JSON.parse(value);
  } catch {
    return fallback;
  }
};

const normalizePromptRow = (row) => {
  if (!row) return null;
  return {
    ...row,
    tags: decodeJson(row.tags_json, []),
    variables: decodeJson(row.variables_json, []),
  };
};

const createAuditLog = async (db, { actor, departmentCode, entityType, entityId, action, summary, metadata = {} }) => {
  await db.run(
    `INSERT INTO prompt_audit_logs
       (actor_id, actor_username, actor_role, department_code, entity_type, entity_id, action, summary, metadata_json)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
    [
      actor?.id || null,
      actor?.username || null,
      actor?.role || null,
      departmentCode || null,
      entityType,
      entityId || null,
      action,
      summary,
      JSON.stringify(metadata),
    ]
  );
};

const createPromptVersion = async (db, promptId, payload, actor, changeNote) => {
  const versionRow = await db.get('SELECT COALESCE(MAX(version_no), 0) + 1 AS next_no FROM prompt_versions WHERE prompt_id = ?', [
    promptId,
  ]);
  const versionNo = Number(versionRow?.next_no || 1);
  const result = await db.run(
    `INSERT INTO prompt_versions
       (prompt_id, version_no, title, scenario, content, variables_json, tags_json, change_note, created_by, created_by_name)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
    [
      promptId,
      versionNo,
      payload.title,
      payload.scenario,
      payload.content,
      encodeJson(payload.variables),
      encodeJson(payload.tags),
      changeNote || '',
      actor?.id || null,
      actor?.username || null,
    ]
  );
  return { id: result.insertId, versionNo };
};
```

```js
const VARIABLE_RE = /\{\{\s*([^{}\s][^{}]*?)\s*\}\}/g;

export const extractPromptVariables = (content) => {
  const result = [];
  const seen = new Set();
  const text = String(content || '');
  let match;
  while ((match = VARIABLE_RE.exec(text))) {
    const name = String(match[1] || '').trim();
    if (!name || seen.has(name)) continue;
    seen.add(name);
    result.push(name);
  }
  return result;
};

export const renderPromptPreview = (content, values = {}) =>
  String(content || '').replace(VARIABLE_RE, (raw, name) => {
    const key = String(name || '').trim();
    const value = values[key];
    return value === undefined || value === null || value === '' ? raw : String(value);
  });

export const statusLabel = (status) => {
  const key = String(status || '').trim();
  if (key === 'published') return '已发布';
  if (key === 'archived') return '已归档';
  return '草稿';
};
```

```jsx
  const raw = String(text || '').trim()
  if (!raw) return { data: null, raw: '' }
  try {
    return { data: JSON.parse(raw), raw }
  } catch {
    return { data: null, raw }

    ...options,
```
